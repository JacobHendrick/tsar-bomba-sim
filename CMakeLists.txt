cmake_minimum_required(VERSION 3.20)
project(tsar_bomba_sim C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# glad 0.1.x declares an old cmake_minimum_required; CMake 4 needs this to accept it.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "Native Wayland backend (X11 works via XWayland)")
set(GLAD_PROFILE "core" CACHE STRING "" FORCE)
set(GLAD_API "gl=4.3" CACHE STRING "" FORCE)
set(GLAD_REPRODUCIBLE ON CACHE BOOL "Use glad's bundled OpenGL specification" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_Declare(glad GIT_REPOSITORY https://github.com/Dav1dde/glad.git GIT_TAG v0.1.36 GIT_SHALLOW TRUE)
FetchContent_Declare(stb URL https://github.com/nothings/stb/archive/2c980bb59875b0d32144a71867fbdebb2f77cd20.tar.gz)
FetchContent_MakeAvailable(glfw glm glad stb)
add_executable(tsar src/main.cpp)
target_include_directories(tsar SYSTEM PRIVATE ${stb_SOURCE_DIR})
target_link_libraries(tsar PRIVATE glfw glm::glm glad)
target_compile_definitions(tsar PRIVATE SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
if(MSVC)
  target_compile_options(tsar PRIVATE /W4)
else()
  target_compile_options(tsar PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli_help COMMAND tsar --help)
  add_test(NAME cli_missing_value COMMAND tsar --grid)
  add_test(NAME cli_invalid_grid COMMAND tsar --grid 17)
  add_test(NAME cli_invalid_steps COMMAND tsar --steps 0)
  add_test(NAME cli_invalid_time COMMAND tsar --shot nan unused.png)
  set_tests_properties(cli_missing_value cli_invalid_grid cli_invalid_steps cli_invalid_time
    PROPERTIES WILL_FAIL TRUE)
  option(TSAR_GPU_TESTS "Run rendering smoke tests (requires an OpenGL 4.3 display)" OFF)
  if(TSAR_GPU_TESTS)
    foreach(scene IN ITEMS zero early cloud mature preset2)
      add_test(NAME render_${scene} COMMAND ${CMAKE_COMMAND}
        -DTSAR=$<TARGET_FILE:tsar> -DSCENE=${scene} -DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/render_smoke.cmake)
      set_tests_properties(render_${scene} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
    endforeach()
  endif()
endif()
